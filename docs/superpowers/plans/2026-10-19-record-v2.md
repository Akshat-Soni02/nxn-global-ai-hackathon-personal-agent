# Record v2, Plan 1: bench, data bugs, fundamental hooks, capture (Phase 0 + Phase 1)

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or
> superpowers:executing-plans to implement this plan task by task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** make every later claim measurable (a bench), fix the bugs that corrupt data, add the replay hooks record needs,
and make capture record complete facts that are never lost silently. This plan ends at **Checkpoint 1**: Rushil records
two short tasks and the traces are read before any Phase 2 work.

**Architecture:**
- **Phase 0** touches the daemon's `finishRecording`, the drill, the LLM client, the agent loop, the vision describer,
  the player's DevTools layer and the run's ending.
- **Phase 1** touches the extension's capture (content script, a new MAIN-world hook, the background worker), the trace
  schema, the daemon's picture reader, the folder watcher, normalise (only for Trash pairing) and Shadow.app's recorder.

Every task is TDD on synthetic data; Swift is written here and verified on Rushil's Mac.

**Tech stack:** TypeScript 7, vitest 5 (jsdom for page code), zod, Biome, esbuild (extension), Swift (Shadow.app).

**Spec:** `docs/superpowers/specs/2026-10-19-record-v2-design.md`. Read it first; this plan argues from it.

**Plans 2 and 3** (Phase 2, and Phase 3 + 4) are written after Checkpoint 1, from the real traces. The spec makes that
checkpoint a gate, and writing their code now would only encode guesses about the new trace format.

**Deviations from the spec in this plan, on purpose:**
- **Fixtures:** Plan 1 creates two fixtures (timesheet, kpi-email-to-sheet). YC and Play Store are added in Plan 2, in
  the new trace format.
- **"Before / after" and "what appeared"** (spec 1.17, 1.19) are *computed* in Plan 2's evidence builder. Plan 1
  stores what they need: each picture's tab and its lines with boxes.
- **Link and card names** (spec 1.5) go to a new descriptor field `seenAs` / `seenFrom`, not `name`. Replay's matcher
  scores `name` against Chrome's accessibility name, and a name Chrome doesn't compute would lower every match score.
- **Vision cost** (spec 0.5) is metered on the daemon's meter in Plan 1. Counting it inside the design's own cap comes
  with Plan 3's orchestrator, which owns one meter for the whole design.

```ts
// The bench's yardstick: what a correct workflow does next time, in the user's words, turned into checks on the saved
// workflow (docs/superpowers/specs/2026-10-19-record-v2-design.md, 0.2). One recording's expect.json lists them; a
// known gap is marked expectedToFail, so a run that hits it reads as ⚠, not as a regression.
import { isAction, isLoop, type Skill, triggerOf, walkSteps } from "@shadow/core";
import { z } from "zod";

const common = { why: z.string(), expectedToFail: z.boolean().optional() };
export const Expectation = z.discriminatedUnion("kind", [
  z.object({ kind: z.literal("has_loop"), over: z.string().optional(), ...common }),
  z.object({ kind: z.literal("uses_today"), ...common }),
  // A write the user showed is still in the workflow: an action ("web.type") whose args match `value` (a regex).
  z.object({ kind: z.literal("write_kept"), action: z.string(), value: z.string().optional(), ...common }),
  z.object({ kind: z.literal("no_text_only_target"), ...common }),
  z.object({ kind: z.literal("trigger"), type: z.enum(["manual", "schedule", "folder_watch"]), ...common }),
  z.object({ kind: z.literal("max_questions"), n: z.number().int().min(0), ...common }),
  // No question asked of the user matched this (a regex, case-insensitive).
  z.object({ kind: z.literal("not_asked"), pattern: z.string(), ...common }),
]);
export type Expectation = z.infer<typeof Expectation>;
export const Expectations = z.array(Expectation);

export interface Verdict {
  expectation: Expectation;
  ok: boolean;
  detail: string;
}

export function checkExpectations(skill: Skill, asked: string[], expectations: Expectation[]): Verdict[] {
  const all = [...walkSteps(skill.steps)].map(({ step }) => step);
  const actions = all.filter(isAction);
  return expectations.map((e): Verdict => {
    switch (e.kind) {
      case "has_loop": {
        const loops = all.filter(isLoop);
        const hit = loops.find((l) => !e.over || l.over.includes(e.over));
        return { expectation: e, ok: Boolean(hit), detail: loops.map((l) => `${l.id} over ${l.over}`).join(", ") || "no loop" };
      }
      case "uses_today": {
        const hit = actions.find((a) => JSON.stringify(a.args).includes("{{today"));
        return { expectation: e, ok: Boolean(hit), detail: hit ? hit.id : "no step uses {{today}}" };
      }
      case "write_kept": {
        const pattern = e.value ? new RegExp(e.value) : undefined;
        const hit = actions.find(
          (a) => `${a.channel}.${a.action}` === e.action && (!pattern || pattern.test(JSON.stringify(a.args))),
        );
        return { expectation: e, ok: Boolean(hit), detail: hit ? hit.id : `no ${e.action} like that` };
      }
      case "no_text_only_target": {
        // A target replay can score: a role, a name, attributes or a selector. Text, near and label alone it can't.
        const weak = actions.filter(
          (a) =>
            a.target &&
            !a.target.role &&
            !a.target.name &&
            Object.keys(a.target.attrs ?? {}).length === 0 &&
            a.target.fallbacks.length === 0,
        );
        return { expectation: e, ok: weak.length === 0, detail: weak.map((a) => a.id).join(", ") || "none" };
      }
      case "trigger": {
        const when = triggerOf(skill).when;
        return { expectation: e, ok: when.some((w) => w.type === e.type), detail: when.map((w) => w.type).join(", ") };
      }
      case "max_questions":
        return { expectation: e, ok: asked.length <= e.n, detail: `${asked.length} asked` };
      case "not_asked": {
        const pattern = new RegExp(e.pattern, "i");
        const hit = asked.find((q) => pattern.test(q));
        return { expectation: e, ok: !hit, detail: hit ?? "not asked" };
      }
    }
  });
}
```

```ts
// design agent moves recorded steps into loops, so a top-level search missed them ("no step s3").
function locate(
  steps: Record<string, unknown>[],
  id: string,
): { list: Record<string, unknown>[]; index: number } | undefined {
  for (let i = 0; i < steps.length; i++) {
    const step = steps[i];
    if (step?.id === id) return { list: steps, index: i };
    for (const key of ["steps", "else"]) {
      const inner = step?.[key];
      if (Array.isArray(inner)) {

        if (found) return found;
      }
    }
  }
  return undefined;
}
```

```ts
export class CdpTimeout extends Error {}

// Navigating, printing and taking a picture may take longer than a query.
const LONG = /^Page\.(navigate|captureScreenshot|printToPDF)$/;

// Every DevTools call gets a limit. Without one, a call Chrome never answered held a step until the daemon gave up
// (35-50 s, seen in three real runs: reports/oct19-two-recordings.md §5). Accessibility queries hung for a page in a
// window behind another one, so a hung accessibility query brings the automation window forward (onSlow) and is tried
// once more.
export function withTimeouts(
  cdp: Cdp,
  options: { ms?: number; longMs?: number; onSlow?: () => Promise<void> } = {},
): Cdp {
  const ms = options.ms ?? 15_000;
  const longMs = options.longMs ?? 30_000;
  const once = <T>(method: string, params?: Record<string, unknown>): Promise<T> => {
    const limit = LONG.test(method) ? longMs : ms;
    let timer: ReturnType<typeof setTimeout> | undefined;
    const timeout = new Promise<never>((_, reject) => {
      timer = setTimeout(
        () => reject(new CdpTimeout(`Chrome did not answer ${method} in ${Math.round(limit / 1000)} s`)),
        limit,
      );
    });
    return Promise.race([cdp.send<T>(method, params), timeout]).finally(() => clearTimeout(timer));
  };
  return {
    async send<T>(method: string, params?: Record<string, unknown>): Promise<T> {
      try {
        return await once<T>(method, params);
      } catch (error) {
        if (!(error instanceof CdpTimeout) || !method.startsWith("Accessibility.") || !options.onSlow) throw error;
        await options.onSlow();
        return once<T>(method, params);
      }
    },
  };
}
```

```ts
const HEADING = "h1, h2, h3, h4, h5, h6, [role=heading]";

// What a person would call a control the accessibility tree leaves nameless: its title, an image's alt text, a heading
// inside it or in its card (the YC job card's link wraps only the logo; the company name is the card's heading), its
// text, or the last part of its address. For the design agent; replay matches Chrome's own name only.
function seen(el: Element): Pick<ElementDescriptor, "seenAs" | "seenFrom"> {
  const title = clean(el.getAttribute("title"));
  if (title) return { seenAs: title, seenFrom: "title" };
  const alt = clean(el.querySelector("img[alt]")?.getAttribute("alt"));
  if (alt) return { seenAs: alt, seenFrom: "alt" };
  const own = clean(el.querySelector(HEADING)?.textContent);
  if (own) return { seenAs: own, seenFrom: "heading" };
  for (let a = el.parentElement, depth = 0; a && depth < 3 && a !== el.ownerDocument.body; a = a.parentElement, depth++) {
    const heading = clean(a.querySelector(HEADING)?.textContent);
    if (heading) return { seenAs: heading, seenFrom: "card" };
  }
  const text = FIELDS.has(el.localName) ? undefined : clean(el.textContent);
  if (text) return { seenAs: text, seenFrom: "text" };
  const last = el.getAttribute("href")?.split(/[?#]/)[0]?.split("/").filter(Boolean).pop();
  if (!last) return {};
  try {
    return { seenAs: decodeURIComponent(last).replace(/[-_]+/g, " "), seenFrom: "href" };
  } catch {
    return { seenAs: last, seenFrom: "href" };
  }
}
```

```ts
const ITEM_ROLES = new Set(["listitem", "row", "article", "gridcell", "option", "treeitem", "menuitem"]);
// Classes that mark state, not shape: the selected job card is still one of the job cards.
const STATE_CLASS = /active|selected|hover|focus|open|current|checked|first|last|odd|even/i;

// Where an element sits in a list: the nearest ancestor (or itself) that is one of 3+ siblings of the same shape (a
// job card among job cards, an email row among rows), its place and the list. "The top job" needs this: before, the
// only trace of it was a selector's :nth-of-type, which replay never reads.
export function listPosition(el: Element): ListPosition | undefined {
  const doc = el.ownerDocument;
  let item: Element | null = el;
  for (let depth = 0; item?.parentElement && depth < 6; item = item.parentElement, depth++) {
    const parent: Element = item.parentElement;
    if (parent === doc.body || parent === doc.documentElement) return undefined;
    const shape = shapeOf(item);
    const same = Array.from(parent.children).filter((c) => shapeOf(c) === shape);
    if (same.length >= 3 && (ITEM_ROLES.has(getRole(item) ?? "") || item.children.length > 0)) {
      return {
        container: describeElement(parent),
        index: same.indexOf(item),
        count: same.length,
        itemText: clean(item.textContent, 120),
      };
    }
  }
  return undefined;
}

function shapeOf(el: Element): string {
  const classes = Array.from(el.classList)
    .filter((c) => !GENERATED.test(c) && !STATE_CLASS.test(c))
    .sort()
    .join(".");
  return `${el.localName}.${classes}.${getRole(el) ?? ""}`;
}
```

```ts
// What a click or an Enter did (spec 1.3), watched for 700 ms after it: did the page go somewhere, did something expand
// or open, did a field empty (a chat message sent), and how much changed near the target. Pages like Gmail re-render
// all the time, so "how much changed" is counted beyond what changed in the same place in the 700 ms before: a click on
// the header of an open email changes nothing, and is noise; a click that opens an email changes a lot near it.
import type { Effect } from "@shadow/core";

export type { Effect };
export const WATCH_MS = 700;
const OPENED: Record<string, Effect["opened"]> = {
  dialog: "dialog",
  alertdialog: "dialog",
  menu: "menu",
  menubar: "menu",
  listbox: "listbox",
};
const OPENABLE = "[role=dialog],[role=alertdialog],[role=menu],[role=listbox],dialog";

interface Change {
  t: number;
  node: Node;
  opened?: Effect["opened"];
}

export interface EffectWatcher {
  // Watches what follows an action on `el`; `done` gets the effect after WATCH_MS, unless the page went away first
  // (then the navigation the browser reports is the effect).
  watch(el: Element, done: (effect: Effect) => void): void;
  stop(): void;
}

export function watchEffects(doc: Document, now: () => number = () => performance.now()): EffectWatcher {
  const changes: Change[] = [];
  const observer = new MutationObserver((records) => {
    const t = now();
    for (const r of records) {
      for (const node of r.addedNodes) changes.push({ t, node, opened: openedBy(node) });
      if (r.type !== "childList" || r.removedNodes.length > 0) changes.push({ t, node: r.target });
    }
    // Two seconds are enough: the 700 ms before the oldest watch, and the watch itself.
    while (changes.length > 0 && (changes[0]?.t ?? t) < t - 2_000) changes.shift();
    if (changes.length > 5_000) changes.splice(0, changes.length - 5_000);
  });
  if (doc.documentElement) {
    observer.observe(doc.documentElement, { subtree: true, childList: true, attributes: true, characterData: true });
  }
  const timers = new Set<ReturnType<typeof setTimeout>>();

  return {
    watch(el, done) {
      const t0 = now();
      const scope = scopeOf(el);
      const near = (c: Change) => scope.contains(c.node) || Boolean(c.opened);
      const before = changes.filter((c) => c.t >= t0 - WATCH_MS && c.t < t0 && near(c)).length;
      const url = doc.location?.href;
      const expandedBefore = expandedOf(el);
      const valueBefore = valueOf(el);
      const timer = setTimeout(() => {
        timers.delete(timer);
        if (!el.ownerDocument.defaultView) return;
        const after = changes.filter((c) => c.t >= t0);
        const extra = Math.max(0, after.filter(near).length - before);
        const effect: Effect = { changed: extra <= 2 ? "none" : extra <= 25 ? "small" : "large" };
        if (doc.location?.href !== url) effect.navigated = true;
        if (expandedBefore === false && expandedOf(el) === true) effect.expanded = true;
        const opened = after.find((c) => c.opened)?.opened;
        if (opened) effect.opened = opened;
        if (valueBefore.trim() && !valueOf(el).trim()) effect.emptied = true;
        done(effect);
      }, WATCH_MS);
      timers.add(timer);
    },
    stop() {
      observer.disconnect();
      for (const timer of timers) clearTimeout(timer);
      timers.clear();
    },
  };
}

function openedBy(node: Node): Effect["opened"] {
  if (!(node instanceof Element)) return undefined;
  const role = node.getAttribute("role") ?? (node.localName === "dialog" ? "dialog" : "");
  if (OPENED[role]) return OPENED[role];
  const inner = node.querySelector(OPENABLE);
  return inner ? openedBy(inner) : undefined;
}

function expandedOf(el: Element): boolean | undefined {
  const holder = el.closest("[aria-expanded]");
  if (holder) return holder.getAttribute("aria-expanded") === "true";
  const details = el.closest("details");
  return details ? details.open : undefined;
}

function valueOf(el: Element): string {
  if ("value" in el && typeof (el as HTMLInputElement).value === "string") return (el as HTMLInputElement).value;
  return el.closest("[contenteditable]:not([contenteditable=false])")?.textContent ?? "";
}

// "Near the target": the closest region it sits in (a dialog, a list item, a section), so churn elsewhere on the page
// doesn't count for it.
function scopeOf(el: Element): Element {
  return el.closest("[role=dialog],dialog,[role=main],main,article,section,li,tr,form") ?? el.parentElement ?? el;
}
```

```ts
// Keys worth recording beyond Enter and Escape (spec 1.8): shortcuts with ⌘, Ctrl or Alt (⌘Enter sends in Gmail), and
// single keys outside text fields (Gmail's e archives, c composes). Never a keylogger: a single key is held for a
// second, and three printable keys within a second are typing (a page that draws its own text has no text field), so
// none of them is recorded. Google Docs, Sheets and Slides record no single keys at all.
export type Modifier = "cmd" | "shift" | "option" | "ctrl";
const EDITING = new Set(["a", "c", "v", "x", "z", "y"]);
const MODIFIER_KEYS = new Set(["Meta", "Control", "Alt", "Shift"]);
export const CANVAS_EDITORS = /^https:\/\/docs\.google\.com\/(spreadsheets|document|presentation)\//;

export function modifiersOf(e: { metaKey: boolean; ctrlKey: boolean; altKey: boolean; shiftKey: boolean }): Modifier[] {
  const mods: Modifier[] = [];
  if (e.metaKey) mods.push("cmd");
  if (e.ctrlKey) mods.push("ctrl");
  if (e.altKey) mods.push("option");
  if (e.shiftKey) mods.push("shift");
  return mods;
}

// A shortcut to record at once: a key with ⌘, Ctrl or Alt that isn't plain editing (copy, paste, cut, select all,
// undo, redo), which the copy and paste events already say.
export function shortcut(key: string, mods: Modifier[]): boolean {
  if (MODIFIER_KEYS.has(key) || !mods.some((m) => m !== "shift")) return false;
  const editing = mods.length === 1 && (mods[0] === "cmd" || mods[0] === "ctrl") && EDITING.has(key.toLowerCase());
  return !editing;
}

interface Held {
  key: string;
  at: number;
  el: Element;
  timer: ReturnType<typeof setTimeout>;
}

export class SingleKeys {
  private held: Held[] = [];
  private burstUntil = Number.NEGATIVE_INFINITY;
  constructor(
    private readonly send: (key: string, at: number, el: Element) => void,
    private readonly waitMs = 1_000,
  ) {}

  // A printable key with no modifier, pressed outside any text field.
  press(key: string, at: number, el: Element): void {
    if (at < this.burstUntil) {
      this.burstUntil = at + this.waitMs; // still typing
      return;
    }
    this.held = this.held.filter((h) => at - h.at < this.waitMs);
    if (this.held.length >= 2) {
      for (const h of this.held) clearTimeout(h.timer);
      this.held = [];
      this.burstUntil = at + this.waitMs;
      return;
    }
    const entry: Held = {
      key,
      at,
      el,
      timer: setTimeout(() => {
        this.held = this.held.filter((h) => h !== entry);
        this.send(key, at, el);
      }, this.waitMs),
    };
    this.held.push(entry);
  }

  stop(): void {
    for (const h of this.held) clearTimeout(h.timer);
    this.held = [];
  }
}
```

```ts
// The tabs a recording used (spec 1.12 and 2.4): for each, the first address it showed, its title, its window, the
// tab that opened it, and whether it was already open when Record was pressed. Replay opens each tab at the right
// place from these. A tab opened while recording has its own tab_open in the trace; a tab that was already open is
// announced once, as a tab_open just before its first event, so only tabs actually used appear.
export interface TabSeen {
  tabId: number;
  windowId?: number;
  url?: string;
  title?: string;
  openerTabId?: number;
  preexisting: boolean;
}
type ChromeTab = { id?: number; windowId?: number; url?: string; title?: string; openerTabId?: number };

export class TabRegistry {
  private tabs = new Map<number, TabSeen>();
  private announced = new Set<number>();

  // At Record: every tab already open.
  start(tabs: ChromeTab[]): void {
    this.tabs.clear();
    this.announced.clear();
    for (const t of tabs) {
      if (t.id === undefined) continue;
      this.tabs.set(t.id, {
        tabId: t.id,
        ...(t.windowId !== undefined ? { windowId: t.windowId } : {}),
        ...(t.url ? { url: t.url } : {}),
        ...(t.title ? { title: t.title } : {}),
        ...(t.openerTabId !== undefined ? { openerTabId: t.openerTabId } : {}),
        preexisting: true,
      });
    }
  }

  // A tab opened while recording: already in the trace as tab_open.
  opened(tab: ChromeTab): void {
    if (tab.id === undefined) return;
    this.tabs.set(tab.id, { tabId: tab.id, windowId: tab.windowId, url: tab.url, openerTabId: tab.openerTabId, preexisting: false });
    this.announced.add(tab.id);
  }

  // The first event from a tab that was open before Record: what to announce first. Undefined after that, for tabs
  // opened while recording, and for tabs never seen.
  firstUse(tabId: number): TabSeen | undefined {
    if (this.announced.has(tabId)) return undefined;
    this.announced.add(tabId);
    return this.tabs.get(tabId);
  }
}
```

`apps/extension/src/shot-queue.ts`:
```ts
// Pictures after events (spec 1.15, 1.16). Chrome takes about two a second (captureVisibleTab), and a newer event used
// to replace a picture still waiting, so a burst kept only its last event's picture and lost the others without a
// word. Now a short queue keeps up to three, dropping the least telling kind first, and whatever can't be taken goes
// in the ledger instead of vanishing.
export interface ShotJob {
  tabId: number;
  eventId: string;
  at: number;
  kind: string;
  // Not before: the page needs a moment to react (400 ms; 1.2 s after a navigation).
  due: number;
  masks?: { x: number; y: number; w: number; h: number }[];
}

const PRIORITY: Record<string, number> = {
  copy: 5,
  select_text: 5,
  click: 4,
  type: 4,
  select: 4,
  check: 4,
  file: 3,
  drop: 3,
  submit: 3,
  key: 3,
  navigate: 2,
  tab_switch: 2,
};
export const GAP_MS = 600;

export class ShotQueue {
  private jobs: ShotJob[] = [];
  constructor(private readonly max = 3) {}

  // Adds a job; returns the one dropped to make room (the least telling, oldest first), for the ledger.
  add(job: ShotJob): ShotJob | undefined {
    this.jobs.push(job);
    if (this.jobs.length <= this.max) return undefined;
    const victim = [...this.jobs].sort((a, b) => (PRIORITY[a.kind] ?? 1) - (PRIORITY[b.kind] ?? 1) || a.at - b.at)[0];
    this.jobs = this.jobs.filter((j) => j !== victim);
    return victim;
  }

  // The job to take now: the oldest one that is due, once the last picture was at least GAP_MS ago.
  next(now: number, lastShot: number): ShotJob | undefined {
    if (now < lastShot + GAP_MS) return undefined;
    const due = this.jobs.filter((j) => j.due <= now).sort((a, b) => a.at - b.at)[0];
    if (due) this.jobs = this.jobs.filter((j) => j !== due);
    return due;
  }

  // How long until a job may be taken, or undefined when there is none.
  wake(now: number, lastShot: number): number | undefined {
    if (this.jobs.length === 0) return undefined;
    return Math.max(0, Math.max(lastShot + GAP_MS, Math.min(...this.jobs.map((j) => j.due))) - now);
  }

  clear(): ShotJob[] {
    const all = this.jobs;
    this.jobs = [];
    return all;
  }
}
```
